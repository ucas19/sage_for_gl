\subsection{degree of atypicality = 2}
\begin{theorem}
  Let $a,b,c,d \in \mathbb{Z}$ and $\lambda = (a,b,c|b,a)$ with $a \neq c, a\neq b$. The projective objects $P_{\lambda}$ in $\mathfrak{B}_{c}$ have the following Verma flag formulae.
\end{theorem}
\begin{proof} The Cases are as follows:\par
  \begin{enumerate}
    \item $a>b>c$

        \item $a>b+1$ \par
        Let $\mu:=\lambda + (0,0,0,1,1) $.
        \begin{equation}\notag
          \begin{aligned}
            P_\mu = M_{a,b,c|b+1,a+1}
          \end{aligned}
        \end{equation}
        \begin{equation}\notag
          \begin{aligned}
            Pr_{\lambda}(P_\mu \otimes V_{3}) &=
M_{a,b,c|b,a}
+ M_{a+1,b,c|b,a+1}
+ M_{a,b+1,c|b+1,a}
+ M_{a+1,b+1,c|b+1,a+1} 
          \end{aligned}
        \end{equation}
All items belong to $P_\lambda$ by \cref{prop:1}.

+  M_{b+2,b,c|b,b+2}
+  M_{b+2,b+2,c|b+2,b+2}\\
                                              &
+  M_{b+3,b+1,c|b+1,b+3}
