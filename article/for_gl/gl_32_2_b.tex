\subsection{degree of atypicality = 2}
\begin{theorem}
  Let $a,b,c,d \in \mathbb{Z}$ and $\lambda = (a,b,c|c,b)$ with $b \neq c$. The projective objects $P_{\lambda}$ in $\mathfrak{B}_{a}$ have the following Verma flag formulae.
\end{theorem}
\begin{proof} The Cases are as follows:\par
\begin{enumerate}
  \item $a>b>c$
    \begin{enumerate}
      \item $b>c+1,a>b+1$ \par
        Let $\mu:=\lambda + (0,0,0,1,1) $.
        \begin{equation}\notag
          \begin{aligned}
            P_\mu = M_{a,b,c|c+1,b+1}
          \end{aligned}
        \end{equation}
        \begin{equation}\notag
          \begin{aligned}
            Pr_{\lambda}(P_\mu \otimes V_{3}) &=
M_{a,b,c|c,b}
+ M_{a,b+1,c|c,b+1}
+ M_{a,b,c+1|c+1,b}
+ M_{a,b+1,c+1|c+1,b+1} 
          \end{aligned}
        \end{equation}
All items belong to $P_\lambda$ by \cref{prop:1}.

      \item $b>c+1,a=b+1$ \par
        Let $\mu:=\lambda + (1,0,0,0,0) $.
        \begin{equation}\notag
          \begin{aligned}
            Pr_{\lambda}(P_\mu \otimes V_{2}) &=
M_{b+1,b,c|c,b}
+ M_{b+1,b+1,c|c,b+1}
+ M_{b+2,b+1,c|c,b+2} \\
                                              &
+ M_{b+1,b,c+1|c+1,b}
+ M_{b+1,b+1,c+1|c+1,b+1}
+ M_{b+2,b+1,c+1|c+1,b+2} \\
          \end{aligned}
        \end{equation}
      Similarly, the set of indeterminate composition factors is $U = \{
M_{b+2,b+1,c+1|c+1,b+2}
\}$. We have $Pr_{\lambda}(P_\mu \otimes V_{2}) = P_\lambda  $ by \cref{tab:verma-analysis_b_1}.

 none of which belongs to the set U.

\begin{equation}\notag
  \begin{aligned}
    \prl(P_\mu \otimes V_8) = Q \oplus P_{b+2,b+1,b+1|b+1,b+2}.
  \end{aligned}
\end{equation}
Since $\prl(P_\mu \otimes V_8)$ is projective (as a summand of the projective module $P_\mu \otimes V$), both $Q$ and $P_\gamma$ would have to be projective. We will show that $Q$ is not projective—thereby contradicting the assumption and eliminating $P_{b+2,b+1,b+1|b+1,b+2}$ as a candidate.

Consequently, $P_\lambda$ is the only possible indecomposable projective summand of $\prl(P_\mu \otimes V_8)$, and we conclude:
\begin{equation}\notag
  \begin{aligned}
\prl(P_\mu \otimes V_8) = P_\lambda.
  \end{aligned}
\end{equation}

Now, we shall show that $Q$ cannot be projective by contradiction.
Suppose, to the contrary, that $Q$ is projective. Then $pr_\nu(Q \otimes V_3)$ is also projective, and its standard filtration admits a decomposition into indecomposable projective summands.
Let $\nu = (b+1,b+1,c+3|c+4,c+2)$. We project $Q \otimes V_4$ on to the $\nu$ linkage block as follows.
\begin{equation}\notag
  \begin{aligned}
    pr_\nu(Q \otimes V_4) &= 
2M_{b+1,b+3,b+1|b+4,b+2}
+2M_{b+1,b+3,b+1|b+2,b+4}
+2M_{b+1,b+1,b+3|b+4,b+2}\\
                                              &
+M_{b+3,b+1,b+1|b+4,b+2}
+2M_{b+1,b+1,b+3|b+2,b+4}
+M_{b+3,b+1,b+1|b+2,b+4}
  \end{aligned}
\end{equation}
Observe that the minimal weight among all composition factors in the standard filtration of $pr_\nu(Q \otimes V_4)$ is $\nu$. By \cref{lemma:2}, this implies that $pr_\nu(Q \otimes V_3)$ must contain at least two copies of $P_\nu$ as direct summands. However, a direct computation shows that
  \begin{equation}\notag
    \begin{aligned}
      P_\nu &=
M_{b+1,b+3,b+1|b+4,b+2}
+M_{b+1,b+3,b+1|b+2,b+4}
+M_{b+1,b+1,b+3|b+4,b+2}\\
                                              &
+M_{b+3,b+1,b+1|b+4,b+2}
+M_{b+1,b+1,b+3|b+2,b+4}
+M_{b+3,b+1,b+1|b+2,b+4}
    \end{aligned}
  \end{equation}
Consequently, $P_\nu$ cannot appear with multiplicity $\geq 2$, yielding a contradiction. Therefore, $Q$ cannot be projective.

\begin{equation}\notag
  \begin{aligned}

  \end{aligned}
\end{equation}

\end{equation}

Consequently, $P_\nu$ cannot appear, yielding a contradiction. Therefore, $Q$ cannot be projective.
