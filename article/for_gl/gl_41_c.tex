\begin{theorem} \label{thm:gl41-c}
  Let $a,b,c,d \in \mathbb{Z}$ and $\lambda = (a,b,c,d|c)$ with $c \notin \{a,b\}$. The projective objects $P_{\lambda}$ in $\mathfrak{B}_{a,b,d}$ have the following Verma flag formulae.
\end{theorem}
\begin{proof} The Cases are as follows:\par
\begin{enumerate}
  \item $ d \neq c $
    \begin{enumerate}
      \item $a \neq c+1, b\neq c+1$
        \begin{enumerate}
          \item $a < c,  c+1 < b < d  $ \par
        Let $\mu:=\lambda + (0,0,0,0|1) $.
        \begin{equation}\notag
          \begin{aligned}
            P_\mu = 2M_{b,d,a,c|c+1}
+ 2M_{d,b,c,a|c+1}
+ M_{b,a,c,d|c+1}
+ M_{d,c,a,b|c+1}
+ M_{a,b,d,c|c+1}
+ M_{c,d,b,a|c+1}
+ M_{c,b,a,d|c+1}
+ M_{a,d,c,b|c+1}
+ M_{d,a,b,c|c+1}
+ M_{b,c,d,a|c+1}
+ 2M_{d,b,a,c|c+1}
+ 2M_{b,d,c,a|c+1}
+ M_{a,b,c,d|c+1}
+ M_{c,d,a,b|c+1}
+ M_{b,a,d,c|c+1}
+ M_{d,c,b,a|c+1}
+ M_{b,c,a,d|c+1}
+ M_{d,a,c,b|c+1}
+ M_{a,d,b,c|c+1}
+ M_{c,b,d,a|c+1}
          \end{aligned}
        \end{equation}
        \begin{equation}\notag
          \begin{aligned}
            Pr_{\lambda}(P_\mu \otimes V_{1}) &=
2M_{b,d,a,c+1|c+1}
+ 2M_{b,d,a,c|c}
+ 2M_{d,b,c+1,a|c+1}
+ 2M_{d,b,c,a|c}
+ M_{b,a,c+1,d|c+1}
+ M_{b,a,c,d|c}
+ M_{d,c+1,a,b|c+1}
+ M_{d,c,a,b|c}
+ M_{a,b,d,c+1|c+1}
+ M_{a,b,d,c|c}
+ M_{c+1,d,b,a|c+1}
+ M_{c,d,b,a|c}
+ M_{c+1,b,a,d|c+1}
+ M_{c,b,a,d|c}
+ M_{a,d,c+1,b|c+1}
+ M_{a,d,c,b|c}
+ M_{d,a,b,c+1|c+1}
+ M_{d,a,b,c|c}
+ M_{b,c+1,d,a|c+1}
+ M_{b,c,d,a|c}
+ 2M_{d,b,a,c+1|c+1}
+ 2M_{d,b,a,c|c}
+ 2M_{b,d,c+1,a|c+1}
+ 2M_{b,d,c,a|c}
+ M_{a,b,c+1,d|c+1}
+ M_{a,b,c,d|c}
+ M_{c+1,d,a,b|c+1}
+ M_{c,d,a,b|c}
+ M_{b,a,d,c+1|c+1}
+ M_{b,a,d,c|c}
+ M_{d,c+1,b,a|c+1}
+ M_{d,c,b,a|c}
+ M_{b,c+1,a,d|c+1}
+ M_{b,c,a,d|c}
+ M_{d,a,c+1,b|c+1}
+ M_{d,a,c,b|c}
+ M_{a,d,b,c+1|c+1}
+ M_{a,d,b,c|c}
+ M_{c+1,b,d,a|c+1}
+ M_{c,b,d,a|c} \\
                                               &=
\sum M_{a,b,c,d|c}
+ \sum M_{b,d,a,c|c}
+ \sum M_{a,b,c+1,d|c+1}
+ \sum M_{b,d,a,c+1|c+1}
          \end{aligned}
        \end{equation}
      Similarly, the set of indeterminate composition factors is $U = \{
M_{b,d,a,c+1|c+1},
M_{b,d,a,c|c},
M_{d,b,c+1,a|c+1},
M_{d,b,c,a|c},

M_{d,b,a,c|c},
M_{b,d,c+1,a|c+1},
M_{b,d,c,a|c}
\}$. 

根据 \cref{tab:verma-analysis_41_c_1}, 仅仅由\cref{prop:1}不足以判断 $P_{b,d,a,c|c}$ 或$P_{d,b,a,c|c}$或$P_{b,d,c,a|c}$或$P_{d,b,c,a|c}$ 是否出现在$\pr_\lambda(P_\lambda \otimes V_{1})$的直和分解中.

为了排除这种可能，我们仅需证明$M_{b,d,a,c|c},M_{d,b,a,c|c},M_{b,d,c,a|c},M_{d,b,c,a|c}$在$\Filt(P_{\lambda})$中的重数是$2$即可。

\begin{equation}\notag
  \begin{aligned}
  \pr_\nu ( ( \pr_\lambda(P_\mu \otimes V_{1}) ) \otimes V_2 ) =
  2P_\nu = 4M_{b,d,a,c|c+1} + 4M_{d,b,a,c|c+1} +4M_{b,d,c,a|c+1} + 4M_{d,b,c,a|c+1}+ ...
  \end{aligned}
\end{equation}

注意到$M_\nu$ 在$\pr_\nu ( ( \pr_\lambda(P_\mu \otimes V_{1}) ) \otimes V_2 )$极小并且重数为 $2$, 且$\nu$只能由$ (a,b,c,d|c) + (0,0,0,0|1)$和$ (a,b,c+1,d|c+1) + (0,0,-1,0|0)$得到.
由\cref{prop:1}, $M_{(a,b,c+1,d|c+1)}\subset \Filt(P_\lambda)$.
因此$\pr_\nu(P_\lambda \otimes V_2) = 2P_\nu$。又由于$(b,d,a,c|c+1)$只能由$(b,d,a,c|c) + (0,0,0,0|1)$和$(b,d,a,c+1|c+1) + (0,0,0,-1|0)$ 得到, 因此我们可以得到$\Filt(P_{\lambda})$中必然有两个$M_{b,d,a,c|c}$，否则$\pr_\nu(P_\lambda \otimes V_2)$不可能包含完整的两个$\Filt(P_\nu)$. 同理，$M_{d,b,a,c|c},M_{b,d,c,a|c},M_{d,b,c,a|c}$在$\Filt(P_{\lambda})$中的重数也是$2$。

综上$Pr_{\lambda}(P_\mu \otimes V_{1}) = P_\lambda  $.
