\subsection{degree of atypicality = 1}

\begin{theorem}
  Let $a,b,c \in \mathbb{Z}$ and $\lambda = (a,b|c,a)$ with $b \neq c$. The projective objects $P_{\lambda}$ in $\mathfrak{B}_{b|c}$ have the following Verma flag formulae.
\end{theorem}
\begin{proof} The Cases are as follows:\par
\begin{enumerate}
  \item $ b \neq a,\; c \neq a $
    \begin{enumerate}
      \item $ b \neq a+1 $ \par 
        Let $\mu:=\lambda + (0,0|0,1) = (a,b|c,a+1) $.
        \begin{equation}\notag
          \begin{aligned}
            Pr_{\lambda}(P_\mu \otimes V_{1}) &=
\sum M_{a,b|c,a}
+ \sum M_{a+1,b|c,a+1}
          \end{aligned}
        \end{equation}
All items belong to $P_\lambda$ by \cref{prop:1}.

      \item $ b = a+1,\; c \neq a+1 $ \par 
        Let $\mu:=\lambda + (1,0|0,0) = (a+1,a+1|c,a) $.
        \begin{equation}\notag
          \begin{aligned}
            Pr_{\lambda}(P_\mu \otimes V_{2}) &=
\sum M_{a,a+1|c,a}
+ \sum M_{a+1,a+1|c,a+1}
          \end{aligned}
        \end{equation}
All items belong to $P_\lambda$ by \cref{prop:1}.

    \end{enumerate}

  \item $ b \neq a,\; c = a $ \par
      Let $\mu:=\lambda + (1,0|0,0) = (a+1,b|a,a) $.
      \begin{equation}\notag
        \begin{aligned}
          Pr_{\lambda}(P_\mu \otimes V_{2}) &=
\sum M_{a,b|a,a}
+ \sum M_{a+1,b|a,a+1}
        \end{aligned}
      \end{equation}
All items belong to $P_\lambda$ by \cref{prop:1}.


  \item $ b = a,\; c \neq a $ \par
      Let $\mu:=\lambda + (0,0|0,1) = (a,a|c,a+1) $.
      \begin{equation}\notag
        \begin{aligned}
          Pr_{\lambda}(P_\mu \otimes V_{1}) &=
\sum M_{a,a|c,a}
+ \sum M_{a,a+1|c,a+1}
        \end{aligned}
      \end{equation}
All items belong to $P_\lambda$ by \cref{prop:1}.

  \item $ b = a,\; c = a $ \par
    Let $\mu:=\lambda + (0,0|1,1) = (a,a|a+1,a+1)$.
      \begin{equation}\notag
        \begin{aligned}
          Pr_{\lambda}(P_\mu \otimes V_{3}) &=
M_{a,a|a,a}
+ M_{a+1,a|a,a+1}
+ M_{a,a+1|a,a+1}
+ M_{a+1,a|a+1,a}
+ M_{a,a+1|a+1,a}
+ M_{a+1,a+1|a+1,a+1} \\
                                               &=
 M_{a,a|a,a}
+ \sum M_{a,a+1|a+1,a}
+  M_{a+1,a+1|a+1,a+1}
        \end{aligned}
      \end{equation}
All items belong to $P_\lambda$ by \cref{prop:1}.
\end{enumerate}
\end{proof}
