\subsection{degree of atypicality = 2}
\begin{theorem}
  Let $a,b,c,d \in \mathbb{Z}$ and $\lambda = (a,b,c|a,b)$ with $a \neq c$. The projective objects $P_{\lambda}$ in $\mathfrak{B}_{c}$ have the following Verma flag formulae.
\end{theorem}
\begin{proof} The Cases are as follows:\par

        \begin{equation}\notag
          \begin{aligned}
            P_\mu = M_{a,b,c|a+1,b+1}
+ M_{a,b,c|b+1,a+1}
          \end{aligned}
        \end{equation}
        \begin{equation}\notag
          \begin{aligned}
            Pr_{\lambda}(P_\mu \otimes V_{3}) &=
M_{a,b,c|a,b}
+ M_{a,b+1,c|a,b+1}
+ M_{a+1,b,c|a+1,b}\\
                                              &
+ M_{a+1,b+1,c|a+1,b+1}
+ M_{a,b,c|b,a}
+ M_{a+1,b,c|b,a+1}\\
                                              &
+ M_{a,b+1,c|b+1,a}
+ M_{a+1,b+1,c|b+1,a+1} \\
                                               &=
\sum M_{a,b,c|a,b}
+ \sum M_{a,b+1,c|a,b+1}
+ \sum M_{a+1,b,c|a+1,b}
+ \sum M_{a+1,b+1,c|a+1,b+1}

        \end{equation}

\begin{equation}\notag
  \begin{aligned}

M_{a+1,b+1,c|a+1,b+1},
M_{a+1,b+1,c|b+1,a+1}
\}.
  \end{aligned}
\end{equation}
