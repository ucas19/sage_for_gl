  Pr_{\lambda}(P_\mu \otimes V_1) &=
M_{a,b|a,a+1}
+M_{a+1,b|a+1,a+1}
+M_{a+2,b|a+1,a+2} \\
                                  &+
M_{b,a|a,a+1}
+M_{b,a+1|a+1,a+1}
+M_{b,a+2|a+1,a+2}\\
                                  &=
\sum M_{a,b|a,a+1}
+\sum M_{a+1,b|a+1,a+1}
+\sum M_{a+2,b|a+1,a+2}
  \end{aligned}
\end{equation}
All items belong to $P_\lambda$ by \cref{prop:1}.

\item $c=a+1,b=c+1$
   Let $\mu:=\lambda + (0,0,1,1) $. 
\begin{equation}\notag
  \begin{aligned}
      P_{\mu} =M_{a+2,a+2|a,a+1}
  \end{aligned}
\end{equation}
\begin{equation}\notag
  \begin{aligned}
  Pr_{\lambda}(P_\mu \otimes V_1) &=
M_{a+2,a+2|a+1,a+2}
+M_{a+1,a+2|a+1,a+1}\\
                                  &+
M_{a+2,a+1|a+1,a+1}
+M_{a,a+2|a,a+1}
+M_{a+2,a|a,a+1} \\
                                  &=
\sum M_{a,a+2|a,a+1}
+\sum M_{a+1,a+2|a+1,a+1}
+ M_{a+2,a+2|a+1,a+2}
  \end{aligned}
\end{equation}
All items belong to $P_\lambda$ by \cref{prop:1}.

  \end{enumerate}
  
\item $c>b>a$
  \begin{enumerate}
    \item $b>a+1$
        Let $\mu:=\lambda + (1,0,0,0) $. 
\begin{equation}\notag
  \begin{aligned}
      P_{\mu} =M_{a+1,b|a,c}
+M_{b,a+1|a,c}
  \end{aligned}
\end{equation}
\begin{equation}\notag
  \begin{aligned}
  Pr_{\lambda}(P_\mu \otimes V_2) &=
M_{a,b|a,c}
+M_{a+1,b|a+1,c}
+M_{b,a|a,c}
+M_{b,a+1|a+1,c}
                                  =
\sum M_{a,b|a,c}
+\sum M_{a+1,b|a+1,c}
  \end{aligned}
\end{equation}
All items belong to $P_\lambda$ by \cref{prop:1}.
    
\item $b=a+1$
        Let $\mu:=\lambda + (1,0,0,0) $. 
\begin{equation}\notag
  \begin{aligned}
      P_{\mu} =M_{a+1,a+1|a,c}
  \end{aligned}
\end{equation}
\begin{equation}\notag
  \begin{aligned}
  Pr_{\lambda}(P_\mu \otimes V_2) &=
+M_{a,a+1|a,c}
+M_{a+1,a|a,c}
+M_{a+1,a+1|a+1,c}
                                  =
\sum M_{a,a+1|a,c}
+ M_{a+1,a+1|a+1,c}
  \end{aligned}
\end{equation}
All items belong to $P_\lambda$ by \cref{prop:1}.
  \end{enumerate}

\end{enumerate}
\end{proof}
