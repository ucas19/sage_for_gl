\begin{theorem} \label{thm:gl41-b}
  Let $a,b,c,d \in \mathbb{Z}$ and $\lambda = (a,b,c,d|b)$ with $a \neq b$. The projective objects $P_{\lambda}$ in $\mathfrak{B}_{a,c,d}$ have the following Verma flag formulae.
\end{theorem}

  \item $a \neq b+1$

  \begin{aligned}

  \end{aligned}
\end{equation}

注意到$M_\nu$ 在$\pr_\nu ( ( \pr_\lambda(P_\mu \otimes V_{2}) ) \otimes V_1 )$极小并且重数为 $1$, 且$\nu$只能由$ (a,b,c,d|b) + (0,0,0,0|-1)$得到.因此$\pr_\nu(P_\lambda \otimes V_1) = P_\nu$。又由于$(b,d,a,c|b-1)$只能由$(b,d,a,c|b) + (0,0,0,0|-1)$得到, 因此我们可以得到$\Filt(P_{\lambda})$中必然有两个$M_{b,d,a,c|b}$，否则$\pr_\nu(P_\lambda \otimes V_1)$不可能包含完整的$\Filt(P_\nu)$. 同理，$M_{d,b,c,a|b}),M_{d,b,a,c|b}, M_{b,d,c,a|b}$在$\Filt(P_{\lambda})$中的重数也是$2$。


情况1: $c = b-1$

\begin{equation}\notag
  \begin{aligned}

  2P_\nu = 2M_\nu + 4M_{b,d,a,c|b+1} + 4M_{b,d,c,a|b+1} + 4M_{d,b,a,c|b+1} + 4M_{d,b,c,a|b+1}  + ...
  \end{aligned}
\end{equation}

注意到$M_\nu$ 在$\pr_\nu ( ( \pr_\lambda(P_\mu \otimes V_{2}) ) \otimes V_2 )$极小并且重数为 $2$, 且$\nu$只能由$ (a,b,c,d|b) + (0,0,0,0|1)$和$ (a,b+1,c,d|b+1) + (0,-1,0,0|0)$ 得到. 由\cref{prop:1}可知，$M_{a,b,c,d|b}$和$M_{a,b+1,c,d|b+1}$都在$\Filt(P_\lambda)$中,
因此$\pr_\nu(P_\lambda \otimes V_2) = 2P_\nu$。

又由于$(b,d,a,c|b+1)$只能由$(b,d,a,c|b) + (0,0,0,0|1)$和$(b+1,d,a,c|b+1) + (-1,0,0,0|0)$得到, 因此我们可以得到$\Filt(P_{\lambda})$中必然有两个$M_{b,d,a,c|b}$和两个$M_{b+1,d,a,c|b+1}$，否则$\pr_\nu(P_\lambda \otimes V_1)$不可能包含完整的两个$\Filt(P_\nu)$. 同理，$M_{d,b,c,a|b}),M_{d,b,a,c|b}, M_{b,d,c,a|b}$在$\Filt(P_{\lambda})$中的重数也是$2$。

综和情况1,2, $Pr_{\lambda}(P_\mu \otimes V_{2}) = P_\lambda  $.
