\begin{theorem}
  Let $a,b,c,d \in \mathbb{Z}$ and $\lambda = (a,b,c|c,a)$ with $a \neq b, c\neq b,a \neq c$. The projective objects $P_{\lambda}$ in $\mathfrak{B}_{b}$ have the following Verma flag formulae.
\end{theorem}
\begin{proof} The Cases are as follows:\par

        \begin{equation}\notag
          \begin{aligned}
            P_\mu = M_{a,b,c|c+1,a+1}
          \end{aligned}
        \end{equation}
        \begin{equation}\notag
          \begin{aligned}
            Pr_{\lambda}(P_\mu \otimes V_{3}) &=
M_{a,b,c|c,a}
+ M_{a+1,b,c|c,a+1}
+ M_{a,b,c+1|c+1,a}
+ M_{a+1,b,c+1|c+1,a+1} 
          \end{aligned}
        \end{equation}
