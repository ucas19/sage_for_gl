\subsection{degree of atypicality = 2}
\begin{theorem}
  Let $a,b,c,d \in \mathbb{Z}$ and $\lambda = (a,b,c|a,c)$ with $a \neq b, c\neq b$. The projective objects $P_{\lambda}$ in $\mathfrak{B}_{b}$ have the following Verma flag formulae.
\end{theorem}
\begin{proof} The Cases are as follows:\par

        \item $b>c+1$ \par
        Let $\mu:=\lambda + (0,0,0,1,1) $.
        \begin{equation}\notag
          \begin{aligned}
            P_\mu = M_{a,b,c|a+1,c+1}
+ M_{a,b,c|c+1,a+1}
          \end{aligned}
        \end{equation}
        \begin{equation}\notag
          \begin{aligned}
            Pr_{\lambda}(P_\mu \otimes V_{3}) &=
M_{a,b,c|a,c}
+ M_{a,b,c+1|a,c+1}
+ M_{a+1,b,c|a+1,c} \\
                                              &
+ M_{a+1,b,c+1|a+1,c+1}
+ M_{a,b,c|c,a}
+ M_{a+1,b,c|c,a+1}\\
                                              &
+ M_{a,b,c+1|c+1,a}
+ M_{a+1,b,c+1|c+1,a+1} \\
                                               &=
\sum M_{a,b,c|a,c}
+ \sum M_{a,b,c+1|a,c+1}
+ \sum M_{a+1,b,c|a+1,c}\\
                                              &
+ \sum M_{a+1,b,c+1|a+1,c+1}
          \end{aligned}
        \end{equation}
All items belong to $P_\lambda$ by \cref{prop:1}.
