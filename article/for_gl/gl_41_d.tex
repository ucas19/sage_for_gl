\begin{theorem} \label{thm:gl41-d}
  Let $a,b,c,d \in \mathbb{Z}$ and $\lambda = (a,b,c,d|d)$ with $d \notin \{a,b,c\}$. The projective objects $P_{\lambda}$ in $\mathfrak{B}_{a,b,c}$ have the following Verma flag formulae.
\end{theorem}
\begin{proof} The Cases are as follows:\par
\begin{enumerate}
  \item $a \neq d+1, b \neq d+1, c \neq d+1$
    \begin{enumerate}
      \item $The generic case: none of the special configurations below occur.$ $not(a \leq c < b < d),(b < a < d, c > d+1)$ \par
        Let $\mu:=\lambda + (0,0,0,0|1) = (a,b,c,d|d+1) $.
        \begin{equation}\notag
          \begin{aligned}
            Pr_{\lambda}(P_\mu \otimes V_{1}) &=
\sum M_{a,b,c,d|d}
+ \sum M_{a,b,c,d+1|d+1}
          \end{aligned}
        \end{equation}
All items belong to $P_\lambda$ by \cref{prop:1}.


      \item $ a < c < b < d $ \par
        Let $\mu:=\lambda + (0,0,0,0|1) $.
        \begin{equation}\notag
          \begin{aligned}
            P_\mu = 2M_{b,d,a,c|d+1}
+ 2M_{d,b,c,a|d+1}
+ M_{b,a,c,d|d+1}
+ M_{d,c,a,b|d+1}
+ M_{a,b,d,c|d+1}
+ M_{c,d,b,a|d+1}
+ M_{c,b,a,d|d+1}
+ M_{a,d,c,b|d+1}
+ M_{d,a,b,c|d+1}
+ M_{b,c,d,a|d+1}
+ 2M_{d,b,a,c|d+1}
+ 2M_{b,d,c,a|d+1}
+ M_{a,b,c,d|d+1}
+ M_{c,d,a,b|d+1}
+ M_{b,a,d,c|d+1}
+ M_{d,c,b,a|d+1}
+ M_{b,c,a,d|d+1}
+ M_{d,a,c,b|d+1}
+ M_{a,d,b,c|d+1}
+ M_{c,b,d,a|d+1}
          \end{aligned}
        \end{equation}
        \begin{equation}\notag
          \begin{aligned}
            Pr_{\lambda}(P_\mu \otimes V_{1}) &=
2M_{b,d+1,a,c|d+1}
+ 2M_{b,d,a,c|d}
+ 2M_{d+1,b,c,a|d+1}
+ 2M_{d,b,c,a|d}
+ M_{b,a,c,d+1|d+1}
+ M_{b,a,c,d|d}
+ M_{d+1,c,a,b|d+1}
+ M_{d,c,a,b|d}
+ M_{a,b,d+1,c|d+1}
+ M_{a,b,d,c|d}
+ M_{c,d+1,b,a|d+1}
+ M_{c,d,b,a|d}
+ M_{c,b,a,d+1|d+1}
+ M_{c,b,a,d|d}
+ M_{a,d+1,c,b|d+1}
+ M_{a,d,c,b|d}
+ M_{d+1,a,b,c|d+1}

+ M_{b,c,d,a|d}
+ 2M_{d+1,b,a,c|d+1}

+ 2M_{b,d+1,c,a|d+1}

          \end{aligned}
        \end{equation}

\begin{equation}\notag
  \begin{aligned}
  \pr_\nu ( ( \pr_\lambda(P_\mu \otimes V_{1}) ) \otimes V_2 ) =
  2P_\nu =2\nu + 4M_{b,d,a,c|d+1} + 4M_{d,b,a,c|d+1} +4M_{b,d,c,a|d+1} + 4M_{d,b,c,a|d+1}+ ...
  \end{aligned}
\end{equation}

注意到$M_\nu$ 在$\pr_\nu ( ( \pr_\lambda(P_\mu \otimes V_{1}) ) \otimes V_2 )$极小并且重数为 $2$, 且$\nu$只能由$ (a,b,c,d|d) + (0,0,0,0|1)$和$ (a,b,c,d+1|d+1) + (0,0,0,-1|0)$得到.
由\cref{prop:1}, $M_{(a,b,c,d+1|d+1)}\subset \Filt(P_\lambda)$.
因此$\pr_\nu(P_\lambda \otimes V_2) = 2P_\nu$。又由于$(b,d,a,c|d+1)$只能由$(b,d,a,c|d) + (0,0,0,0|1)$和$(b,d+1,a,c|d+1) + (0,-1,0,0|0)$ 得到, 因此我们可以得到$\Filt(P_{\lambda})$中必然有两个$M_{b,d,a,c|d}$，否则$\pr_\nu(P_\lambda \otimes V_2)$不可能包含完整的两个$\Filt(P_\nu)$. 同理，$M_{d,b,a,c|d},M_{b,d,c,a|d},M_{d,b,c,a|d}$在$\Filt(P_{\lambda})$中的重数也是$2$。

综上$Pr_{\lambda}(P_\mu \otimes V_{1}) = P_\lambda  $.
